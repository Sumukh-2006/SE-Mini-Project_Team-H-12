cmake_minimum_required(VERSION 3.20)
project(HMS VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(hms_core src/database.cpp src/auth.cpp src/patients.cpp src/doctors.cpp src/appointments.cpp)
target_include_directories(hms_core PUBLIC include)
target_link_libraries(hms_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
target_compile_options(hms_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hms_app src/main.cpp)
target_link_libraries(hms_app PRIVATE hms_core)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(hms_tests tests/core_tests.cpp)
  target_link_libraries(hms_tests PRIVATE hms_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(hms_tests)
endif()
